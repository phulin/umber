\def\etexextensions{
  \immediate\write16{UMBER-ETEX26-EXTENSIONS-EXTENDED}
  \def\extensioneffect{EXTENDED}
  \message{VERSION=\the\eTeXversion\eTeXrevision}
  \def\ordinarymacro{ORDINARY}
  \protected\def\protectedmacro{PROTECTED}
  \edef\protectedcopy{A\protectedmacro B}
  \edef\unexpandedcopy{C\unexpanded{\ordinarymacro D}E}
  \edef\detokenizedcopy{\detokenize{\ordinarymacro F G}}
  \message{PROTECTED-COPY=\protectedcopy}
  \message{UNEXPANDED-COPY=\unexpandedcopy}
  \message{DETOKENIZED-COPY=\detokenizedcopy}
  \everyeof={\message{EVERYEOF-REPLAY}}
  \scantokens{\message{SCANTOKENS-BODY}}
  \openin1=extensions-readline.txt
  \readline1 to\readlinecopy
  \closein1
  \message{READLINE=\meaning\readlinecopy}
  \message{NUMEXPR=\the\numexpr 2+3*4\relax}
  \message{DIMEXPR=\the\dimexpr 1pt+2pt\relax}
  \message{GLUEEXPR=\the\glueexpr 1pt plus 2pt+3pt\relax}
  \message{MUEXPR=\the\muexpr 1mu plus 2mu+3mu\relax}
  \message{GLUE-ENQUIRIES=
    \the\gluestretch 1pt plus 2fil minus 3fill,
    \the\glueshrink 1pt plus 2fil minus 3fill,
    \number\gluestretchorder 1pt plus 2fil minus 3fill,
    \number\glueshrinkorder 1pt plus 2fil minus 3fill}
  \message{GLUE-CONVERSIONS=
    \the\mutoglue 1mu plus 2mu,
    \the\gluetomu 1pt plus 2pt}
  \ifdefined\undefinedprobe
    \message{IFDEFINED-WRONG}
  \else
    \message{IFDEFINED-FALSE}
  \fi
  \ifcsname undefinedprobe\endcsname
    \message{IFCSNAME-WRONG}
  \else
    \message{IFCSNAME-FALSE}
  \fi
  \iffontchar\nullfont 0
    \message{IFFONTCHAR-WRONG}
  \else
    \message{IFFONTCHAR-FALSE}
  \fi
  \unless\ifdefined\undefinedprobe
    \message{UNLESS-TRUE}
  \else
    \message{UNLESS-WRONG}
  \fi
  \begingroup
    \message{GROUP=
      \number\currentgrouplevel,\number\currentgrouptype}
    \iftrue
      \message{CONDITION=
        \number\currentiflevel,\number\currentiftype,
        \number\currentifbranch}
    \fi
  \endgroup
  \message{INTERACTION=\number\interactionmode}
  \setbox0=\hbox{}
  \message{LASTNODE=\number\lastnodetype}
  \message{FONTCHAR=
    \the\fontcharwd\nullfont 0,
    \the\fontcharht\nullfont 0,
    \the\fontchardp\nullfont 0,
    \the\fontcharic\nullfont 0}
  \parshape=2 1pt 2pt 3pt 4pt
  \message{PARSHAPE=
    \the\parshapeindent 1,\the\parshapelength 1,
    \the\parshapedimen 3}
  \count300=11
  \dimen300=12pt
  \skip300=13pt plus 2fil
  \muskip300=14mu plus 3mu
  \toks300={SPARSE-TOKENS}
  \setbox300=\hbox{}
  \global\count301=15
  \tracingassigns=1
  \tracinggroups=1
  \tracingifs=1
  \tracingscantokens=1
  \tracingnesting=1
  \predisplaydirection=-1
  \TeXXeTstate=1
  \lastlinefit=500
  \savingvdiscards=1
  \savinghyphcodes=1
}
\expandafter\ifx\csname eTeXversion\endcsname\relax
  \message{UMBER-ETEX26-EXTENSIONS-COMPATIBILITY}
  \def\extensioneffect{COMPATIBILITY}
\else
  \etexextensions
\fi
\immediate\openout1=extensions-effects.out
\immediate\write1{EXTENSION-EFFECT=\extensioneffect}
\immediate\closeout1
\shipout\hbox{}
\end
